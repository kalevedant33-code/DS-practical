Practical 11: PRIORITY QUEUE OF STUDENTS BASED ON CGPA

MAX = 10
queue = []
processed = 0

while True:
    print("\n1. Insert student")
    print("2. Process highest CGPA student")
    print("3. Display queue")
    print("4. Show students processed")
    print("5. Exit")

    choice = input("Enter choice: ")

    if choice == "1":
        if len(queue) == MAX:
            print("Queue is full.")
        else:
            roll = input("Enter roll number: ")
            name = input("Enter name: ")
            cgpa = float(input("Enter CGPA: "))

            student = [roll, name, cgpa]
            position = 0

            # Find where to insert; equal CGPAs stay in arrival order
            while position < len(queue) and queue[position][2] >= cgpa:
                position += 1

            queue.insert(position, student)
            print("Student inserted.")

            for item in queue:
                print(item)

    elif choice == "2":
        if len(queue) == 0:
            print("Queue is empty.")
        else:
            student = queue.pop(0)
            print("Processing:", student)
            processed += 1

            print("Students processed:", processed)
            if queue:
                for item in queue:
                    print(item)
            else:
                print("Queue is empty.")

    elif choice == "3":
        if len(queue) == 0:
            print("Queue is empty.")
        else:
            for item in queue:
                print(item)

    elif choice == "4":
        print("Students processed:", processed)

    elif choice == "5":
        break

    else:
        print("Invalid choice.")
