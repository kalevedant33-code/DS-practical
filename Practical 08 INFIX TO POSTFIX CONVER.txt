Practical 8: INFIX TO POSTFIX CONVERSION AND EVALUATION OF A MARKS FORMULA

# Convert infix to postfix
def convert(expression):
    stack = []
    postfix = ""

    for ch in expression.replace(" ", ""):
        if ch.isalpha():                 # A, B, C or D
            postfix += ch
        elif ch == "(":
            stack.append(ch)
        elif ch == ")":
            while stack[-1] != "(":
                postfix += stack.pop()
            stack.pop()                  # Remove "("
        else:                            # Operator
            while stack and stack[-1] != "(":
                postfix += stack.pop()
            stack.append(ch)

    while stack:
        postfix += stack.pop()

    return postfix


# Evaluate postfix expression
def evaluate(postfix, values):
    stack = []

    for ch in postfix:
        if ch.isalpha():
            stack.append(values[ch])
        else:
            b = stack.pop()
            a = stack.pop()

            if ch == "+":
                stack.append(a + b)
            elif ch == "-":
                stack.append(a - b)
            elif ch == "*":
                stack.append(a * b)
            elif ch == "/":
                stack.append(a / b)
            elif ch == "^":
                stack.append(a ** b)

    return stack.pop()


# Given formula and values
formula = "(A+B)/D+C"
values = {"A": 18, "B": 22, "D": 2, "C": 8}

postfix = convert(formula)
answer = evaluate(postfix, values)

print("Infix:", formula)
print("Postfix:", " ".join(postfix))
print("Result:", answer)
