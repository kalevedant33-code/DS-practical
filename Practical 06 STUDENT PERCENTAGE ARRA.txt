Practical 6: STUDENT PERCENTAGE ARRANGEMENT USING QUICK SORT

def partition(percentages, low, high):
    pivot = percentages[high]
    i = low - 1

    for j in range(low, high):
        if percentages[j] >= pivot:
            i += 1
            percentages[i], percentages[j] = percentages[j], percentages[i]

    percentages[i + 1], percentages[high] = percentages[high], percentages[i + 1]
    return i + 1


def quick_sort(percentages, low, high):
    if low < high:
        pivot_index = partition(percentages, low, high)
        quick_sort(percentages, low, pivot_index - 1)
        quick_sort(percentages, pivot_index + 1, high)


n = int(input("Enter number of students: "))
percentages = []

for i in range(n):
    percentage = float(input(f"Enter percentage of student {i + 1}: "))
    percentages.append(percentage)

quick_sort(percentages, 0, n - 1)

print("Percentages in descending order:")
print(percentages)
