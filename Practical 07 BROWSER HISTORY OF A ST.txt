Practical 7: BROWSER HISTORY OF A STUDENT PORTAL USING A STACK

MAX = 5
history = []
top = -1

while True:
    print("\n1. Visit page")
    print("2. Go back")
    print("3. Show current page")
    print("4. Show history")
    print("5. Exit")

    choice = input("Enter choice: ")

    if choice == "1":
        if top == MAX - 1:
            print("Stack Overflow: History is full.")
        else:
            url = input("Enter page URL: ")
            history.append(url)
            top += 1
            print("Visited:", url)

    elif choice == "2":
        if top <= 0:
            print("Cannot go back. Home page remains open.")
        else:
            print("Closed:", history.pop())
            top -= 1
            print("Current page:", history[top])

    elif choice == "3":
        if top == -1:
            print("History is empty.")
        else:
            print("Current page:", history[top])

    elif choice == "4":
        if top == -1:
            print("History is empty.")
        else:
            print("Browsing history (current to oldest):")
            for i in range(top, -1, -1):
                print(history[i], "<-- Current" if i == top else "")

    elif choice == "5":
        break

    else:
        print("Invalid choice.")
